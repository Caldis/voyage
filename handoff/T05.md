```glsl
  if (inBezel > 0.0 && isMain > 0.5) {
  // 机翼（wing-shading.glsl.ts）：a 是覆盖率，< 1 的是轮廓上的像素（边缘已在里面超采样），要和窗外按覆盖率混合
  vec4 wing = wingView(ro, rd, (PANE_DEPTH - ro.z) / rd.z, sunC, eSkyH, eDown, belowAlbedo, cloud);
  view = vec3(0.0);
  if (wing.a < 1.0) {
    vec3 rdW = uCabinToWorld * rd;
    view = outsideRadiance(rdW, cloud);
    // 远处的飞机和航迹云在云层之上，挡在海面和云前面
    vec4 tr = trafficRadiance(rdW);

```
